# StreamShield Linux dependencies

# Core dependencies are system GObject Introspection packages that must be
# installed via the system package manager, not pip:
#
#   Ubuntu/Debian:
#     sudo apt install \
#       python3-gi python3-gi-cairo gir1.2-gtk-4.0 \
#       gir1.2-atspi-2.0 at-spi2-core \
#       gir1.2-ayatanaappindicator3-0.1
#
#   Fedora:
#     sudo dnf install \
#       python3-gobject gtk4 \
#       at-spi2-core at-spi2-atk \
#       libayatana-appindicator-gtk3
#
#   Arch:
#     sudo pacman -S \
#       python-gobject gtk4 \
#       at-spi2-core at-spi2-atk \
#       libayatana-appindicator
#
# The packages below are pip-installable wrappers/stubs. On most distros,
# the system packages above are sufficient and these are not needed.
